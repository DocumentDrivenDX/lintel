from __future__ import annotations

import hashlib
import json
import re
from pathlib import Path

from jsonschema import Draft202012Validator

root = Path('/private/tmp/lintel-api4-rich-build27-20261105/relay-client')
artifact = json.loads(Path('/private/tmp/lintel-api4-rich-build27-final-20261105/output/prepared-api-v4-rich-weft7-proof.json').read_bytes())
schema = json.loads(Path('/Users/erik/Projects/lintel/docs/helix/02-design/schemas/client-manifest.schema.json').read_bytes())
operation_source = root / 'src/connection_operations.ts'
document_sha = hashlib.sha256(operation_source.read_bytes()).hexdigest()
operation_names = (
    'RichCollectionsQuery',
    'RichNotesPaginationQuery',
    'RichTasksPaginationQuery',
    'RichNodeLookupQuery',
    'RichNodesLookupQuery',
)
operations = [
    {
        'name': name,
        'kind': 'query',
        'documentSha256': document_sha,
        'artifact': f'src/__generated__/{name}.graphql.ts',
    }
    for name in operation_names
]
files = []
for path in sorted(p for p in root.rglob('*') if p.is_file() and p.name != 'client-manifest.json'):
    relative = path.relative_to(root).as_posix()
    files.append({'path': relative, 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()})
manifest = {
    'format': 'lintel.client/1',
    'application': artifact['application'],
    'identityNamespace': artifact['identityNamespace'],
    'apiDigest': artifact['apiDigest'],
    'schemaDigest': artifact['schemaDigest'],
    'toolchain': {'relay': '21.0.1', 'typescript': '6.0.3', 'bun': '1.4.3'},
    'profiles': {'graphql': 'lintel.graphql/1', 'execution': 'lintel.execution/1', 'scalars': 'lintel.scalars/1'},
    'operations': operations,
    'files': files,
}
Draft202012Validator.check_schema(schema)
errors = list(Draft202012Validator(schema).iter_errors(manifest))
if errors:
    raise SystemExit('\n'.join(error.message for error in errors))
(root / 'client-manifest.json').write_text(json.dumps(manifest, sort_keys=True, separators=(',', ':')) + '\n')
for name in operation_names:
    generated = (root / f'src/__generated__/{name}.graphql.ts').read_text()
    match = re.search(r'"text": "(.*?)"\n\s*\}', generated, re.DOTALL)
    if not match:
        raise AssertionError(f'missing compiled operation text: {name}')
    text = json.loads('"' + match.group(1) + '"')
    if any(directive in text for directive in ('@connection', '@refetchable', '@argumentDefinitions', '@arguments')):
        raise AssertionError(f'compiler-only directive leaked in wire text: {name}')
print('manifest_schema_validation=passed')
print('manifest_sha256', hashlib.sha256((root / 'client-manifest.json').read_bytes()).hexdigest())
print('manifest_files', len(files))
print('manifest_operations', json.dumps(operation_names))
print('document_sha256', document_sha)
print('compiled_wire_texts=5; compiler_directives_absent=true')
print('generated_ts_decimal_string', 'DecimalString' in (root / 'src/__generated__/RichNotesFragment.graphql.ts').read_text())
print('schema_digest_matches', hashlib.sha256((root / 'schema.graphql').read_bytes()).hexdigest() == artifact['schemaDigest'])
