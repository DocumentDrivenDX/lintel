class BlockBuildAndUmf(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname: str, path: object = None, target: object = None):
        if (
            fullname == 'umf'
            or fullname.startswith('umf_')
            or fullname.startswith('umf.')
            or fullname.startswith('lintel_build')
        ):
            raise ModuleNotFoundError(f'blocked runtime import: {fullname}')
        return None


sys.meta_path.insert(0, BlockBuildAndUmf())
import lintel
from lintel.graphql import create_router
from lintel.node_id import PreparedRoots, StringKeyNodeCodec
from lintel.ports import (
    ConnectionPage,
    Edge,
    ElementRef,
    NodeKey,
    NodeValue,
    RequestScope,
)
from lintel.projection import TrustedPreparedRef, load_prepared_api

ROOT = Path('/Users/erik/Projects/lintel')
raw = (ROOT / 'tests/fixtures/prepared-api-v4-weft7-build27-proof.json').read_bytes()
trust = json.loads(
    (ROOT / 'tests/fixtures/prepared-api-v4-weft7-build27-trust.json').read_bytes()
)
pin = TrustedPreparedRef(
    artifact_sha256=trust['artifactSha256'],
    artifact_digest=trust['artifactDigest'],
    application=trust['application'],
    api_digest=trust['apiDigest'],
    exposure_sha256=trust['exposureSha256'],
    sources=tuple((row['identity'], row['sha256']) for row in trust['sources']),
    capabilities=tuple(tuple(row) for row in trust['capabilities']),
    binding_profile=trust['bindingProfile'],
    scalar_codec_profile=trust['scalarCodecProfile'],
    limits=tuple(tuple(row) for row in trust['limits']),
    generated_roots_sha256=trust['generatedRootsSha256'],
    generated_connections_sha256=trust['generatedConnectionsSha256'],
)
prepared = load_prepared_api(raw, pin)


def validate_key(
    _module: str, _record: str, _key_id: str, components: tuple[object, ...]
) -> tuple[object, ...]:
    return tuple(cast(dict[str, object], item)['string'] for item in components)


codec = StringKeyNodeCodec.for_prepared_api(
    cast(PreparedRoots, prepared), {(ElementRef('demo', 'Note'), 'primary'): validate_key}
)


class FakePort:
    def capabilities(self):
        return dict(prepared.capabilities)

    @asynccontextmanager
    async def session_planned(self, scope, intents):
        assert scope.application == 'fixture-app' and len(intents) == 1
        yield self

    @asynccontextmanager
    async def session(self, scope):
        yield self

    async def planned_connection(self, intent):
        return ConnectionPage(
            (
                Edge(
                    'edge:note-a',
                    NodeValue(
                        NodeKey(
                            ElementRef('demo', 'Note'), 'primary', ({'string': 'a'},)
                        ),
                        {ElementRef('demo', 'owner'): {'string': 'A'}},
                    ),
                ),
            ),
            False,
            False,
        )

    async def get_many(self, *args):
        raise AssertionError('unexpected Node query')

    async def connection(self, *args):
        raise AssertionError('unexpected connection dispatch')

    async def related_many(self, *args):
        raise AssertionError('unexpected relationship dispatch')

    async def invoke(self, *args):
        raise AssertionError('unexpected command dispatch')

    async def observe(self, *args):
        raise AssertionError('unexpected observation dispatch')


async def resolve_scope(_request, digest):
    return RequestScope(
        'fixture-app', digest, 'query', object(), 'api4-wheel-probe', monotonic() + 60
    )


app = FastAPI()
app.include_router(create_router(prepared, FakePort(), codec, resolve_scope))
with TestClient(app) as client:
    response = client.post(
        '/graphql',
        json={

        },
        headers={'content-type': 'application/json', 'x-lintel-api': prepared.api_digest},
    )
    print('status', response.status_code)
    print('body', json.dumps(response.json(), sort_keys=True, separators=(',', ':')))
print('lintel_origin', lintel.__file__)
